Ansible for infrastructure automation

Streamlit for the web interface

Ollama LLM (phi3:mini) for AI reasoning

🧪 Playbook 1 — Connection Validator

This playbook verifies connectivity to hosts and stores results in JSON artifacts.

connection_validator.yml
---
- name: Validate connectivity to hosts
  hosts: linux
  gather_facts: no

  tasks:

    - name: Ping host
      ping:
      register: ping_result
      ignore_errors: yes

    - name: Prepare result object
      set_fact:
        host_result:
          hostname: "{{ inventory_hostname }}"
          ip: "{{ ansible_host }}"
          success: "{{ ping_result is succeeded }}"
          details: "{{ ping_result.msg | default('pong') }}"

    - name: Append result to list
      set_fact:
        validation_results: "{{ validation_results | default([]) + [host_result] }}"
      run_once: false

- name: Save results to artifact
  hosts: localhost
  gather_facts: no

  vars:
    timestamp: "{{ lookup('pipe','date +%Y%m%d_%H%M%S') }}"

  tasks:

    - name: Save JSON artifact
      copy:
        content: "{{ hostvars | dict2items | map(attribute='value.validation_results') | list | flatten | to_nice_json }}"
        dest: "~/artifacts/raw_runs/connection_validator_{{ timestamp }}.json"
🔍 Playbook 2 — Hostname Comparison

This playbook compares:

hostname defined in Ansible inventory

actual hostname on the server

check_server_name.yml
---
- name: Compare inventory hostname with real hostname
  hosts: linux
  gather_facts: yes

  tasks:

    - name: Get real hostname
      command: hostname
      register: hostname_cmd
      ignore_errors: yes

    - name: Determine hostname status
      set_fact:
        hostname_status: >-
          {% if hostname_cmd.stdout == inventory_hostname %}
          MATCH
          {% else %}
          MISMATCH
          {% endif %}

    - name: Determine agent suggestion
      set_fact:
        suggestion: >-
          {% if hostname_cmd.stdout == inventory_hostname %}
          No action required
          {% else %}
          Update inventory hostname or rename server
          {% endif %}

    - name: Prepare result
      set_fact:
        hostname_result:
          inventory_hostname: "{{ inventory_hostname }}"
          ip_address: "{{ ansible_host }}"
          real_hostname: "{{ hostname_cmd.stdout }}"
          status: "{{ hostname_status }}"
          agent_suggestion: "{{ suggestion }}"

    - name: Append to results
      set_fact:
        hostname_results: "{{ hostname_results | default([]) + [hostname_result] }}"

- name: Save hostname comparison results
  hosts: localhost
  gather_facts: no

  vars:
    timestamp: "{{ lookup('pipe','date +%Y%m%d_%H%M%S') }}"

  tasks:

    - name: Save JSON artifact
      copy:
        content: "{{ hostvars | dict2items | map(attribute='value.hostname_results') | list | flatten | to_nice_json }}"
        dest: "~/artifacts/raw_runs/check_server_name_{{ timestamp }}.json"

💻 app.py (Full Code)
import streamlit as st
import subprocess
import json
import csv
import os
from datetime import datetime
import glob
import requests

ARTIFACT_DIR = os.path.expanduser("~/artifacts/raw_runs")
CSV_DIR = os.path.expanduser("~/artifacts")

OLLAMA_URL = "http://localhost:11434/api/generate"
MODEL = "phi3:mini"


def cleanup_old_files():

    json_files = sorted(glob.glob(f"{ARTIFACT_DIR}/*.json"), key=os.path.getmtime)
    csv_files = sorted(glob.glob(f"{CSV_DIR}/*.csv"), key=os.path.getmtime)

    if len(json_files) > 1:
        for f in json_files[:-1]:
            os.remove(f)

    if len(csv_files) > 1:
        for f in csv_files[:-1]:
            os.remove(f)


def call_llm(prompt):

    response = requests.post(
        OLLAMA_URL,
        json={
            "model": MODEL,
            "prompt": prompt,
            "stream": False
        }
    )

    return response.json()["response"]


def detect_intent(user_prompt):

    prompt = f"""
Classify the user's request.

Return one of:
FAILED_HOSTS_REPORT
RUN_HOSTNAME_CHECK
UNKNOWN

User request:
{user_prompt}
"""

    result = call_llm(prompt)

    return result.strip()


def load_latest_json():

    files = sorted(glob.glob(f"{ARTIFACT_DIR}/*.json"), key=os.path.getmtime)
    if not files:
        return None

    latest = files[-1]

    with open(latest) as f:
        data = json.load(f)

    return data, latest


def generate_failed_hosts_csv(data):

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    csv_file = f"{CSV_DIR}/failed_hosts_{timestamp}.csv"

    with open(csv_file, "w", newline="") as f:

        writer = csv.writer(f)
        writer.writerow([
            "Hostname",
            "IP Address",
            "Last Playbook Run",
            "Reason of Fail",
            "Suggested Fix"
        ])

        for host in data:

            if not host["success"]:

                analysis_prompt = f"""
Host failed connectivity check.

Details:
{host}

Explain failure and suggest fix.
"""

                analysis = call_llm(analysis_prompt)

                writer.writerow([
                    host["hostname"],
                    host["ip"],
                    datetime.now(),
                    analysis.split("\n")[0],
                    analysis
                ])

    return csv_file


def run_hostname_check():

    subprocess.run(
        "ansible-playbook -i ~/ansible/inventory.ini ~/ansible/check_server_name.yml",
        shell=True
    )


def generate_hostname_csv(data):

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    csv_file = f"{CSV_DIR}/hostname_comparison_{timestamp}.csv"

    with open(csv_file, "w", newline="") as f:

        writer = csv.writer(f)
        writer.writerow([
            "Inventory Hostname",
            "Real Hostname",
            "Status",
            "Agent Suggestion"
        ])

        for host in data:

            writer.writerow([
                host["inventory_hostname"],
                host["real_hostname"],
                host["status"],
                host["agent_suggestion"]
            ])

    return csv_file


cleanup_old_files()

st.set_page_config(page_title="Ansible AI Agent", layout="wide")
st.title("Ansible AI Agent")

user_prompt = st.text_area(
    "Enter instruction for the AI agent"
)

if st.button("Run AI Agent"):

    intent = detect_intent(user_prompt)

    if intent == "FAILED_HOSTS_REPORT":

        data, file = load_latest_json()

        csv_file = generate_failed_hosts_csv(data)

        st.success("Failed hosts report generated")
        st.write(csv_file)

    elif intent == "RUN_HOSTNAME_CHECK":

        run_hostname_check()

        data, file = load_latest_json()

        csv_file = generate_hostname_csv(data)

        st.success("Hostname comparison report generated")
        st.write(csv_file)

    else:

        st.warning("Agent could not understand the request")
